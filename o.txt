def normalize_nft_url(url: str) -> str:
    if not url:
        return ""
    u = str(url).strip()
    m = re.search(r"(?:t\.me|telegram\.me)/nft/([a-zA-Z0-9_-]+)", u, re.IGNORECASE)
    if m:
        return f"https://t.me/nft/{m.group(1)}"
    m2 = re.match(r"^([a-zA-Z0-9_-]+-\d+)$", u)
    if m2:
        return f"https://t.me/nft/{m2.group(1)}"
    return ""


def extract_all_nft_urls(raw_input: Any) -> List[str]:
    if not raw_input:
        return []
    text = " ".join(str(x) for x in raw_input) if isinstance(raw_input, list) else str(raw_input)

    results = []
    for match in re.finditer(r"(?:https?://)?(?:www\.)?(?:t\.me|telegram\.me)/nft/([a-zA-Z0-9_-]+)", text, re.IGNORECASE):
        clean = f"https://t.me/nft/{match.group(1)}"
        if clean not in results:
            results.append(clean)

    if not results:
        for token in re.split(r"[\s,;\n]+", text.strip()):
            norm = normalize_nft_url(token)
            if norm and norm not in results:
                results.append(norm)

    return results


def parse_nft_name(url: str) -> str:
    norm = normalize_nft_url(url) or str(url)
    slug = norm.rstrip("/").split("/")[-1]
    if "-" in slug:
        name_part, num_part = slug.rsplit("-", 1)
        if num_part.isdigit():
            return f"{name_part} #{num_part}"
    return slug or "NFT Gift"


def render_gifts_list_html(urls: List[str]) -> str:
    if len(urls) == 1:
        return f"<a href='{urls[0]}'>{parse_nft_name(urls[0])}</a>"
    return "\n".join([f"• <a href='{u}'>{parse_nft_name(u)}</a>" for u in urls])

def init_db():
    with sqlite3.connect(DB_NAME, timeout=10) as conn:
        conn.execute("PRAGMA journal_mode=WAL;")
        conn.execute("PRAGMA busy_timeout=5000;")
        cursor = conn.cursor()
        cursor.execute("CREATE TABLE IF NOT EXISTS subscriptions (user_id INTEGER PRIMARY KEY, expires_at TIMESTAMP)")
        cursor.execute("CREATE TABLE IF NOT EXISTS business_connections (connection_id TEXT PRIMARY KEY, owner_id INTEGER, can_reply BOOLEAN)")
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS offers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                chat_id INTEGER,
                message_id INTEGER DEFAULT 0,
                business_connection_id TEXT DEFAULT '',
                owner_id INTEGER,
                nft_url TEXT,
                nft_name TEXT,
                price REAL,
                currency TEXT,
                created_at TIMESTAMP,
                status TEXT DEFAULT 'active',
                gift_received INTEGER DEFAULT 0,
                last_checkpoint INTEGER DEFAULT 360
            )
        """)
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS business_chats (
                chat_id INTEGER,
                owner_id INTEGER,
                connection_id TEXT,
                title TEXT,
                username TEXT,
                updated_at TIMESTAMP,
                PRIMARY KEY (chat_id, owner_id)
            )
        """)
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS users (
                user_id INTEGER PRIMARY KEY,
                username TEXT,
                full_name TEXT,
                joined_at TIMESTAMP
            )
        """)
        cursor.execute("CREATE TABLE IF NOT EXISTS bot_config (key TEXT PRIMARY KEY, value TEXT)")
        conn.commit()


def set_config(key: str, value: str):
    with sqlite3.connect(DB_NAME) as conn:
        conn.execute("INSERT OR REPLACE INTO bot_config (key, value) VALUES (?, ?)", (key, str(value)))
        conn.commit()


def get_config(key: str, default: str = None) -> str | None:
    with sqlite3.connect(DB_NAME) as conn:
        row = conn.execute("SELECT value FROM bot_config WHERE key = ?", (key,)).fetchone()
        return row[0] if row else default


def save_user_profile(user: User):
    if not user:
        return
    clean_uname = user.username.replace("@", "").strip() if user.username else ""
    full_name = " ".join(filter(None, [user.first_name, user.last_name])).strip()

    with sqlite3.connect(DB_NAME) as conn:
        conn.execute("""
            INSERT INTO users (user_id, username, full_name, joined_at)
            VALUES (?, ?, ?, ?)
            ON CONFLICT(user_id) DO UPDATE SET
                username=excluded.username,
                full_name=excluded.full_name
        """, (user.id, clean_uname, full_name, datetime.now().isoformat()))
        conn.commit()

    if clean_uname:
        set_config(f"user_uname_{user.id}", clean_uname)
    if full_name:
        set_config(f"user_name_{user.id}", full_name)


def set_user_receiver(user_id: int, username: str):
    clean_username = username.replace("@", "").strip()
    set_config(f"receiver_{user_id}", clean_username)


def get_user_receiver(user_id: int, fallback_username: str = None) -> str:
    saved = get_config(f"receiver_{user_id}")
    if saved and saved.strip():
        return saved.strip()
    if fallback_username and fallback_username.strip():
        clean = fallback_username.replace("@", "").strip()
        set_user_receiver(user_id, clean)
        return clean
    return ""

def get_seller_display(chat_id: int) -> str:
    with sqlite3.connect(DB_NAME) as conn:
        row = conn.execute(
            "SELECT username, title FROM business_chats WHERE chat_id = ? ORDER BY updated_at DESC LIMIT 1",

        ).fetchone()
        if row:
            uname, title = row
            if uname:
                return f"@{uname.replace('@', '').strip()}"
            if title:
                return re.sub(r"\s*\(<code>\d+<\/code>\)|\s*\(\d+\)", "", title).strip()
    return f"Клиент (ID: {chat_id})"

        "custom_decline": get_config(f"style_dec_{user_id}", "")

def apply_user_bold(owner_id: int, text: str) -> str:
    if not owner_id:
        return text
    st = get_user_style(owner_id)
    if st.get("bold", "0") == "1":
        clean_inner = text.replace("<b>", "").replace("</b>", "")
        return f"<b>{clean_inner}</b>"
    return text


def is_admin(user: User) -> bool:
    if not user:
        return False
    if ADMIN_USERNAME and user.username and user.username.lower() == ADMIN_USERNAME.lower():
        set_config("admin_id", str(user.id))
        return True
    return get_config("admin_id") == str(user.id)


async def check_user_subscription(bot: Bot, user_id: int) -> bool:
    if not REQUIRED_CHANNEL:
        return True
    admin_id = get_config("admin_id")
    if admin_id and str(user_id) == str(admin_id):
        return True
    try:
        member = await bot.get_chat_member(chat_id=REQUIRED_CHANNEL, user_id=user_id)
        return member.status in ["creator", "administrator", "member"]
    except Exception as e:
        logging.warning(f"Ошибка проверки подписки на {REQUIRED_CHANNEL}: {e}")
        return True

def add_subscription(user_id: int, days: int) -> datetime:
    with sqlite3.connect(DB_NAME) as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT expires_at FROM subscriptions WHERE user_id = ?", (user_id,))
        row = cursor.fetchone()
        now = datetime.now()
        new_exp = (datetime.fromisoformat(row[0]) if row and datetime.fromisoformat(row[0]) > now else now) + timedelta(days=days)
        cursor.execute("INSERT OR REPLACE INTO subscriptions (user_id, expires_at) VALUES (?, ?)", (user_id, new_exp.isoformat()))
        conn.commit()
        return new_exp


def remove_subscription(user_id: int):
    with sqlite3.connect(DB_NAME) as conn:
        conn.execute("DELETE FROM subscriptions WHERE user_id = ?", (user_id,))
        conn.commit()


def has_active_subscription(user: User) -> bool:
    if is_admin(user):
        return True
    with sqlite3.connect(DB_NAME) as conn:
        row = conn.execute("SELECT expires_at FROM subscriptions WHERE user_id = ?", (user.id,)).fetchone()
        if not row:
            return False
        return datetime.fromisoformat(row[0]) > datetime.now()

def get_owner_chats_encoded(owner_id: int) -> str:
    with sqlite3.connect(DB_NAME) as conn:
        rows = conn.execute("""
            SELECT chat_id, connection_id, title, username 
            FROM business_chats 
            WHERE owner_id = ? 
            ORDER BY updated_at DESC LIMIT 50
        """, (owner_id,)).fetchall()

    chats_data = [{"id": r[0], "cid": r[1], "t": r[2][:22], "u": r[3]} for r in rows]
    raw_json = json.dumps(chats_data, ensure_ascii=False)
    return base64.urlsafe_b64encode(raw_json.encode("utf-8")).decode("utf-8")


def create_offer(chat_id: int, message_id: int, conn_id: str, owner_id: int, urls: List[str], price: float, currency_code: str, created_at: str) -> int:
    stored_urls = ",".join(urls)
    names = ", ".join([parse_nft_name(u) for u in urls])
    with sqlite3.connect(DB_NAME) as conn:
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO offers (chat_id, message_id, business_connection_id, owner_id, nft_url, nft_name, price, currency, created_at, status, gift_received, last_checkpoint)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, 'active', 0, 360)
        """, (chat_id, message_id, conn_id, owner_id, stored_urls, names, price, currency_code, created_at))
        conn.commit()
        return cursor.lastrowid

class BusinessChatTrackerMiddleware(BaseMiddleware):
    async def __call__(self, handler: Callable[[TelegramObject, Dict[str, Any]], Awaitable[Any]], event: TelegramObject, data: Dict[str, Any]) -> Any:
        if isinstance(event, Message) and event.business_connection_id:
            conn_id = event.business_connection_id
            owner_id = get_connection_owner(conn_id)
            if not owner_id and event.from_user:
                owner_id = event.from_user.id
                save_business_connection(conn_id, owner_id, True)

            if owner_id:
                full_name = " ".join(filter(None, [event.chat.first_name, event.chat.last_name]))
                chat_title = full_name or event.chat.title or "Личный чат"
                save_business_chat(event.chat.id, owner_id, conn_id, chat_title, event.chat.username or "")

            has_gift = bool(
                event.gift
                or getattr(event, "unique_gift", None)
                or (getattr(event, "action", None) and hasattr(event.action, "gift"))
            )

            if has_gift and owner_id:
                with sqlite3.connect(DB_NAME) as conn:
                    row = conn.execute("""
                        SELECT id, owner_id, nft_url, price, currency, message_id
                        FROM offers
                        WHERE chat_id = ? AND status = 'active'
                        ORDER BY id DESC LIMIT 1
                    """, (event.chat.id,)).fetchone()

                if row:
                    offer_id, off_owner_id, nft_url_str, price_raw, currency_raw, card_msg_id = row
                    urls = nft_url_str.split(",")
                    receiver_username = get_user_receiver(off_owner_id)

                    all_received, _ = await verify_all_gifts_received(urls, receiver_username, 0)
                    if not all_received:
                        return await handler(event, data)

                    mark_offer_completed(offer_id)

                    price = int(price_raw) if float(price_raw).is_integer() else price_raw
                    curr_display = render_currency(currency_raw)

                    seller_display = get_seller_display(event.chat.id)
                    buyer_display = get_buyer_display(off_owner_id)
                    gifts_list = render_gifts_list_html(urls)

                    deal_done_notify = (
                        f"🎉 <b>NFT подтвержден на профиле {buyer_display}! Сделка закрыта.</b>\n\n"
                        f"Квитанция: <code>#TX-00{offer_id}</code>\n"
                        f"👤 <b>Продавец (клиент):</b> {seller_display}\n"

def get_top_buyers_data(hours: int | None = None, only_today: bool = False, limit: int = 10) -> list:
    with sqlite3.connect(DB_NAME) as conn:
        rows = conn.execute(
            "SELECT owner_id, nft_url, price, currency, created_at FROM offers WHERE status = 'completed'"
        ).fetchall()

    if only_today:
        cutoff = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
    elif hours:
        cutoff = datetime.now() - timedelta(hours=hours)
    else:
        cutoff = None

    buyers = {}
    for owner_id, nft_url, price_raw, currency_raw, created_at_val in rows:
        if not owner_id:
            continue
        if cutoff and created_at_val:
            try:
                c_date = datetime.fromisoformat(str(created_at_val).replace("Z", "").replace(" ", "T").strip())
                if c_date < cutoff:
                    continue
            except Exception:
                pass

        try:
            price = float(str(price_raw).replace(",", ".").strip())
        except (ValueError, TypeError):
            price = 0.0

        nfts_count = len(nft_url.split(",")) if nft_url else 1
        curr_key = render_currency(currency_raw or "star")

        if owner_id not in buyers:
            buyers[owner_id] = {
                "owner_id": owner_id,
                "deals": 0,
                "nfts": 0,
                "spent": 0.0,
                "curr": curr_key
            }
        buyers[owner_id]["deals"] += 1
        buyers[owner_id]["nfts"] += nfts_count
        buyers[owner_id]["spent"] += price

    sorted_buyers = sorted(buyers.values(), key=lambda x: (x["deals"], x["spent"]), reverse=True)
    return sorted_buyers[:limit]


def calculate_today_stats() -> dict:
    with sqlite3.connect(DB_NAME) as conn:
        rows = conn.execute("SELECT status, price, currency, nft_url, created_at FROM offers").fetchall()

    today_start = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
    total_created = 0
    completed_count = 0
    declined_count = 0
    expired_count = 0
    total_nfts_received = 0
    volumes = {"⭐": 0.0, "💎 GRAM": 0.0, "USDT": 0.0}

    for status, price_raw, currency_raw, nft_url, created_at_val in rows:
        if created_at_val:
            try:
                c_date = datetime.fromisoformat(str(created_at_val).replace("Z", "").replace(" ", "T").strip())
                if c_date < today_start:
                    continue
            except Exception:
                pass

        total_created += 1
        try:
            price = float(str(price_raw).replace(",", ".").strip())
        except (ValueError, TypeError):
            price = 0.0

        if status == "completed":
            completed_count += 1
            curr_key = render_currency(currency_raw or "star")
            volumes[curr_key] = volumes.get(curr_key, 0.0) + price
            if nft_url:
                total_nfts_received += len(nft_url.split(","))
        elif status == "declined":
            declined_count += 1
        elif status == "expired":
            expired_count += 1

    conversion = round((completed_count / total_created) * 100, 1) if total_created > 0 else 0.0

    return {
        "total": total_created,
        "completed": completed_count,
        "declined": declined_count,
        "expired": expired_count,
        "nfts": total_nfts_received,
        "conversion": conversion,
        "volumes": volumes
    }

async def verify_all_gifts_received(nft_urls: List[str], expected_username: str, gift_received_flag: int = 0) -> tuple[bool, List[str]]:
    if gift_received_flag == 1:
        return True, []

    clean_user = expected_username.replace("@", "").strip().lower()
    if not clean_user:
        return False, [parse_nft_name(u) for u in nft_urls]

    missing = []
    headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"}

    async with aiohttp.ClientSession(headers=headers) as session:
        for url in nft_urls:
            found = False
            try:
                async with session.get(url, timeout=8) as resp:
                    if resp.status == 200:
                        html = await resp.text()
                        patterns = [
                            rf'href="https?://t\.me/{re.escape(clean_user)}"',
                            rf'@{re.escape(clean_user)}\b',
                            rf'Owner.*?\b{re.escape(clean_user)}\b',
                            rf'Владелец.*?\b{re.escape(clean_user)}\b'
                        ]
                        for pat in patterns:
                            if re.search(pat, html, re.IGNORECASE | re.DOTALL):
                                found = True
                                break
            except Exception:
                pass
            if not found:
                missing.append(parse_nft_name(url))

    return len(missing) == 0, missing

def get_remaining_time(created_at_str: str) -> str:
    if not created_at_str:
        return "5h 59m"
    created_at = datetime.fromisoformat(created_at_str)
    expires_at = created_at + timedelta(hours=OFFER_DURATION_HOURS)
    remaining = expires_at - datetime.now()
    if remaining.total_seconds() <= 0:
        return "0h 0m"
    hours = int(remaining.total_seconds() // 3600)
    minutes = int((remaining.total_seconds() % 3600) // 60)
    return f"{hours}h {minutes}m"


def render_currency(currency_raw: str) -> str:
    if not currency_raw:
        return "⭐"
    curr = str(currency_raw).lower().strip()
    if any(k in curr for k in ["star", "звезд", "⭐", "xtr"]):
        return "⭐"
    elif any(k in curr for k in ["gram", "грам", "ton", "тон", "💎"]):
        return "💎 GRAM"
    elif any(k in curr for k in ["usdt", "usd", "$"]):
        return "USDT"
    return "⭐"

async def offer_expiration_worker(bot: Bot):
    while True:
        await asyncio.sleep(25)
        try:
            with sqlite3.connect(DB_NAME) as conn:
                cursor = conn.cursor()
                cursor.execute("""
                    SELECT id, chat_id, message_id, business_connection_id, owner_id, nft_url, price, currency, created_at, last_checkpoint
                    FROM offers WHERE status = 'active'
                """)
                active_offers = cursor.fetchall()

            now = datetime.now()
            for row in active_offers:
                offer_id, chat_id, message_id, conn_id, owner_id, nft_url_str, price, currency_raw, created_at_str, last_cp = row
                if not created_at_str or not message_id:
                    continue

                created_at = datetime.fromisoformat(created_at_str)
                rem_seconds = (timedelta(hours=OFFER_DURATION_HOURS) - (now - created_at)).total_seconds()
                lang = get_user_lang(owner_id)

                if rem_seconds <= 0:
                    with sqlite3.connect(DB_NAME) as conn:
                        conn.execute("UPDATE offers SET status = 'expired' WHERE id = ?", (offer_id,))
                        conn.commit()
                    try:
                        exp_text = apply_user_bold(owner_id, LANGS.get(lang, LANGS["ru"])["expired_text"])
                        await bot.edit_message_text(
                            chat_id=chat_id,
                            message_id=message_id,
                            business_connection_id=conn_id,
                            text=exp_text,
                            reply_markup=None,
                            link_preview_options=LinkPreviewOptions(is_disabled=True)
                        )
                    except Exception:
                        pass
                    await asyncio.sleep(0.35)
                    continue

                rem_minutes = int(rem_seconds // 60)
                last_cp = last_cp or 360
                triggered_cp = None

                for cp in SILENT_EDIT_CHECKPOINTS:
                    if rem_minutes <= cp < last_cp:
                        triggered_cp = cp
                        break

                if triggered_cp is not None:
                    with sqlite3.connect(DB_NAME) as conn:
                        conn.execute("UPDATE offers SET last_checkpoint = ? WHERE id = ?", (triggered_cp, offer_id))
                        conn.commit()

                    receiver_username = get_user_receiver(owner_id)
                    if not receiver_username:
                        continue

                    urls = nft_url_str.split(",")
                    updated_text = build_offer_card_text(owner_id, receiver_username, urls, price, currency_raw, lang, created_at_str)
                    preview = build_offer_preview_options(owner_id, urls[0])
                    kb = get_offer_keyboard(offer_id, lang, owner_id)

                    try:
                        await bot.edit_message_text(
                            chat_id=chat_id,
                            message_id=message_id,
                            business_connection_id=conn_id,
                            text=updated_text,
                            reply_markup=kb,
                            link_preview_options=preview
                        )
                    except Exception:
                        pass
                    await asyncio.sleep(0.35)
        except Exception as e:
            logging.error(f"Ошибка воркера таймеров: {e}")

                s.expires_at
            FROM users u
            LEFT JOIN subscriptions s ON u.user_id = s.user_id
            ORDER BY u.joined_at DESC
            LIMIT ? OFFSET ?
        """, (PAGE_SIZE, offset)).fetchall()

    total_pages = max(1, (total_count + PAGE_SIZE - 1) // PAGE_SIZE)
    return rows, total_count, total_pages


def build_users_page_message(page: int = 1) -> tuple[str, InlineKeyboardMarkup]:
    rows, total_count, total_pages = get_users_page_data(page)

    lines = [
        f"👥 <b>База пользователей</b> (Всего: <code>{total_count}</code>)",
        f"<i>Страница {page} из {total_pages}</i>\n"
    ]

    now = datetime.now()
    if not rows:
        lines.append("<i>В базе пока нет пользователей.</i>")
    else:
        for idx, (u_id, uname, fname, joined, exp_at) in enumerate(rows, start=(page - 1) * PAGE_SIZE + 1):
            u_title = f"@{uname}" if uname else "<i>Без юзернейма</i>"
            name_part = f" ({fname})" if fname else ""
            date_clean = joined[:16].replace("T", " ") if joined else "—"

            if exp_at and datetime.fromisoformat(exp_at) > now:
                exp_date_str = exp_at[:16].replace("T", " ")
                sub_str = f"Активна (до {exp_date_str}) ✅"
            else:
                sub_str = "Нет ❌"

            lines.append(
                f"<b>{idx}. 👤 {u_title}</b>{name_part}\n"
                f"   ├ 🆔 TG: <code>{u_id}</code>\n"
                f"   ├ 💎 Подписка: <b>{sub_str}</b>\n"
                f"   └ 📅 Регистрация: {date_clean}\n"
            )

    nav_buttons = []
    if page > 1:
        nav_buttons.append(InlineKeyboardButton(text="◀️ Назад", callback_data=f"users_page:{page - 1}"))
    nav_buttons.append(InlineKeyboardButton(text=f"{page} / {total_pages}", callback_data="users_noop"))
    if page < total_pages:
        nav_buttons.append(InlineKeyboardButton(text="Вперёд ▶️", callback_data=f"users_page:{page + 1}"))

    kb = InlineKeyboardMarkup(inline_keyboard=[
        nav_buttons,
        [InlineKeyboardButton(text="📥 Скачать базу (.csv таблица)", callback_data="users_download_csv")]
    ])

    return "\n".join(lines), kb

@router.callback_query(F.data == "users_download_csv")
async def cb_users_download_csv(call: CallbackQuery):
    if not is_admin(call.from_user):
        return

    now = datetime.now()
    with sqlite3.connect(DB_NAME) as conn:
        rows = conn.execute("""
            SELECT 
                u.user_id, 
                u.username, 
                u.full_name, 
                u.joined_at,
                s.expires_at
            FROM users u
            LEFT JOIN subscriptions s ON u.user_id = s.user_id
            ORDER BY u.joined_at DESC
        """).fetchall()

    csv_lines = ["Telegram ID,Username,Full Name,Joined Date,Has Active Sub,Subscription Expiry"]
    for r in rows:
        u_id, uname, fname, joined, exp_at = r
        clean_fname = (fname or "").replace(",", " ").replace('"', '')
        has_sub = "YES" if (exp_at and datetime.fromisoformat(exp_at) > now) else "NO"
        csv_lines.append(
            f'"{u_id}","{uname or ""}","{clean_fname}","{joined or ""}","{has_sub}","{exp_at or ""}"'
        )

    csv_data = "\ufeff" + "\n".join(csv_lines)
    file_bytes = csv_data.encode("utf-8")
    doc = BufferedInputFile(file_bytes, filename=f"users_{datetime.now().strftime('%d_%m_%Y')}.csv")

    await call.message.answer_document(doc, caption=f"📊 Полная база пользователей: <b>{len(rows)} чел.</b>")
    await call.answer()

@router.message(Command("broadcast"))
async def cmd_broadcast(message: Message, command: CommandObject, bot: Bot):
    if not is_admin(message.from_user):
        return

    text_to_send = command.args
    if not text_to_send:
        await message.answer("❌ Формат: <code>/broadcast Ваш текст рассылки</code>")
        return

    with sqlite3.connect(DB_NAME) as conn:
        user_ids = [row[0] for row in conn.execute("SELECT user_id FROM users").fetchall()]

    if not user_ids:
        await message.answer("❌ База пользователей пуста.")
        return

    status_msg = await message.answer(f"⏳ Начинаю рассылку для <b>{len(user_ids)}</b> пользователей...")
    success, failed = 0, 0

    for uid in user_ids:
        try:
            await bot.send_message(
                chat_id=uid,
                text=text_to_send,
                link_preview_options=LinkPreviewOptions(is_disabled=True)
            )
            success += 1
        except Exception:
            failed += 1
        await asyncio.sleep(0.05)

    await status_msg.edit_text(
        f"📢 <b>Рассылка завершена!</b>\n\n"
        f"✅ Успешно доставлено: <b>{success}</b>\n"
        f"❌ Не доставлено: <b>{failed}</b>\n"
        f"👥 Всего в базе: <b>{len(user_ids)}</b>"
    )

@router.message(F.web_app_data)
async def handle_webapp_data(message: Message, bot: Bot):
    user = message.from_user
    save_user_profile(user)
    try:
        data = json.loads(message.web_app_data.data)
        action = data.get("action")

        if "style_preview" in data:
            save_user_style(
                user.id,
                str(data.get("style_preview", "1")),
                str(data.get("style_layout", "row")),
                str(data.get("style_icons", "0")),

                str(data.get("style_template", "default")),
                str(data.get("style_bold", "0")),
                str(data.get("custom_accept", "")),
                str(data.get("custom_decline", ""))
            )

        if action == "save_settings":
            new_receiver = data.get("receiver", "").replace("@", "").strip()
            new_lang = data.get("lang", "ru")
            if new_lang in ["ru", "en"]:
                set_user_lang(user.id, new_lang)
            if not new_receiver:
                await message.answer(NO_USERNAME_WARNING_RU, reply_markup=build_webapp_keyboard(user))
                return
            set_user_receiver(user.id, new_receiver)
            await message.answer(
                f"✅ <b>Настройки и кастомизация сохранены!</b>\n"
                f"Получатель подарков: <b>@{new_receiver}</b>",
                reply_markup=build_webapp_keyboard(user)
            )

        elif action == "send_offer":
            if not await check_user_subscription(bot, user.id):
                await message.answer(
                    f"⚠️ <b>Для отправки офферов требуется подписка на канал {REQUIRED_CHANNEL}!</b>",
                    reply_markup=get_sub_check_keyboard()
                )
                return

            if not has_active_subscription(user):
                await message.answer("У вас нет активной подписки. Введите <code>/tempagifts</code>.")
                return

            raw_target = str(data.get("chat_id", "")).strip()
            conn_id = (data.get("conn_id") or "").strip() or get_owner_default_conn_id(user.id)
            urls = extract_all_nft_urls(data.get("nft_urls") or data.get("nft_url") or "")

            price = float(data.get("price", 0))
            currency = data.get("currency", "star")
            lang = data.get("lang") or get_user_lang(user.id)
            receiver = (data.get("receiver") or "").replace("@", "").strip() or get_user_receiver(user.id, user.username)

            if not receiver:
                await message.answer(NO_USERNAME_WARNING_RU, reply_markup=build_webapp_keyboard(user))
                return
            if not urls:
                await message.answer("❌ Не удалось распознать ссылку на подарок.")
                return
            if not conn_id:
                await message.answer("❌ Подключите бота в настройках Telegram Business.")
                return

            # Проверка получателя
            if raw_target.startswith("@") or not raw_target.lstrip("-").isdigit():
                clean_uname = raw_target.replace("@", "").strip().lower()
                with sqlite3.connect(DB_NAME) as conn:
                    row = conn.execute(
                        "SELECT chat_id, connection_id, title, username FROM business_chats WHERE owner_id = ? AND LOWER(username) = ? LIMIT 1",
                        (user.id, clean_uname)
                    ).fetchone()
                if row:
                    chat_id = int(row[0])
                    conn_id = row[1] or conn_id
                    target_display = f"@{row[3]}" if row[3] else f"{row[2]}"
                else:
                    try:
                        chat_obj = await bot.get_chat(f"@{clean_uname}")
                    except Exception as e:
                        await message.answer(f"❌ Пользователь @{clean_uname} не найден: {e}")
                        return

                    # Защита от отправки в группы и каналы
                    if chat_obj.type != "private":
                        await message.answer(
                            f"❌ <b>Ошибка чата!</b>\n\n"

                            f"Бизнес-офферы можно отправлять только в <b>личные сообщения пользователям</b>."
                        )
                        return

                    chat_id = chat_obj.id
                    full_name = " ".join(filter(None, [chat_obj.first_name, chat_obj.last_name])) or f"@{clean_uname}"
                    save_business_chat(chat_id, user.id, conn_id, full_name, clean_uname)
                    target_display = f"@{clean_uname}"
            else:
                chat_id = int(raw_target)
                target_display = get_seller_display(chat_id)

            await send_offer_card_to_chat(bot, chat_id, conn_id, user.id, receiver, urls, price, currency, lang)
            await send_offer_log_to_bot_dm(bot, user, target_display, receiver, price, currency, urls)

    except Exception as e:
        logging.error(f"Ошибка WebApp: {e}")
        await message.answer(f"❌ Ошибка отправки оффера: {e}")


@router.business_message(F.text.regexp(r"(?is)^!создать\s+(.+)"))
async def handle_create_offer(message: Message, bot: Bot):
    conn_id = message.business_connection_id
    owner = message.from_user
    save_user_profile(owner)
    lang = get_user_lang(owner.id)

    if not await check_user_subscription(bot, owner.id):
        await smart_reply(message, f"⚠️ Для создания офферов требуется подписка на канал {REQUIRED_CHANNEL}!")
        return

    if not has_active_subscription(owner):
        await smart_reply(message, LANGS[lang]["no_sub"])
        return

    receiver_username = get_user_receiver(owner.id, owner.username)
    if not receiver_username:
        await smart_reply(message, NO_USERNAME_WARNING_RU)
        return

    body = re.sub(r"(?i)^!создать\s+", "", message.text.strip()).strip()
    tail_match = re.search(r"(\d+(?:[.,]\d+)?)\s+([a-zA-Zа-яА-Я⭐💎$]+)\s*$", body)

    if not tail_match:
        await smart_reply(message, "❌ Формат: <code>!создать ссылка 500 star</code>")
        return

    price = float(tail_match.group(1).replace(",", "."))
    currency_raw = tail_match.group(2)
    links_part = body[:tail_match.start()].strip()

    urls = extract_all_nft_urls(links_part)
    if not urls:
        await smart_reply(message, "❌ Ссылка не найдена. Пример: <code>!создать t.me/nft/DurovsCap-1 500 star</code>")
        return

    try:
        try:
            await bot.delete_business_messages(business_connection_id=conn_id, message_ids=[message.message_id])
        except Exception:
            pass

        await send_offer_card_to_chat(bot, message.chat.id, conn_id, owner.id, receiver_username, urls, price, currency_raw, lang)

        target_display = f"@{message.chat.username}" if message.chat.username else get_seller_display(message.chat.id)
        await send_offer_log_to_bot_dm(bot, owner, target_display, receiver_username, price, currency_raw, urls)
    except Exception as e:
        await smart_reply(message, f"❌ Ошибка: {e}")

            f"⚠️ <i>Собеседник нажал кнопку отмены в чате.</i>"

@router.message(Command("sub"))
async def cmd_sub(message: Message, command: CommandObject):
    if not is_admin(message.from_user):
        return
    args = command.args.split() if command.args else []
    if len(args) == 2 and args[0].isdigit() and args[1].isdigit():
        exp = add_subscription(int(args[0]), int(args[1]))
        await smart_reply(message, f"✅ Подписка для <code>{args[0]}</code> до {exp.strftime('%d.%m.%Y %H:%M')}")


@router.message(Command("unsub"))
async def cmd_unsub(message: Message, command: CommandObject):
    if not is_admin(message.from_user):
        return
    if command.args and command.args.isdigit():
        remove_subscription(int(command.args))
        await smart_reply(message, f"✅ Подписка <code>{command.args}</code> отключена.")
